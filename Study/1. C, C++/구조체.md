구조체는 "각 원소의 타입이 제각각인 배열"과 같다.
```c
struct Human {
	int age;
	int height;
	int weight;
};
```
이 때문에 배열에서는 배열의 타입만으로 모든 원소의 타입을 알 수 있었지만 구조체는 그렇지 않다.
따라서 구조체는 정의할 때 **모든 원소의 타입을 명시해주어야 한다**.

### 초기화
구조체의 정의 내부에서는 변수를 초기화할 수 없다.
```c
#include <stdio.h>
struct Books {
	char name[30];
	char auth[30];
	char publ[30];
	int borrowed;
};
int main() {
	struct Books book_list[3];
	int i;
	
	for (i = 0; i < 3; i++) {
		scanf("%s%s%s", book_list[i].name, book_list[i].auth, book_list[i].publ)
		book_list[i].borrowed = 0;
	}
	
	return 0;
}
```

```c
struct Books book_list[3];
```
먼저 위 문장을 살펴보자.
`int arr[3]`에서 `int`가 하나의 타입이듯이, `struct Books`가 하나의 타입이라고 생각하면 된다.
그런데, `int arr[3]`을 하면 `arr`에 `int` 형 원소가 3개 만들어지듯이, `book_list` 배열에는 `struct Books` 형의 변수가 3개 만들어지는 것이다.

### 구조체 포인터
```c
/* 구조체 포인터 */
#include <stdio.h>
struct test {
	int a, b;
};
int main() {
	struct test st;
	struct test *ptr;
	
	ptr = &st;
	
	(*ptr).a = 1;
	(*ptr).b = 2;
	
	printf("st 의 a 멤버 : %d \n", st.a);
	printf("st 의 b 멤버 : %d \n", st.b);
	
	return 0;
}
```
위에서 말했듯 `struct test` 역시 하나의 타입이다.
즉, 구조체는 `int` 나 `char` 처럼 한 개의 타입을 창조하는 것과 마찬가지이다. 때문에 `int *`, `char * `처럼 포인터를 사용할 경우에도 `struct test *ptr`로 정의한다.
여기서 주의할 점은 `ptr`은 절대로 구조체가 아니라는 점이다. `ptr` 역시 다른 모든 포인터처럼 4byte의 공간(x64의 경우 8byte)을 차지하는 것이다.

```c
(*ptr).a = 1;
(*ptr).b = 2;
```
멤버에 접근할 경우 `(*ptr).a`와 같이 괄호로 감싸서 접근하게 되는데, 이때는 `ptr->a`를 사용해주는 것이 더 좋다.

### 구조체 안의 구조체
```c
/* 구조체 안의 구조체*/
#include <stdio.h>
struct employee {
	int age;
	int salary;
};
struct company {
	struct employee data;
	char name[10];
};
int main() {
	struct company Kim;
	
	Kim.data.age = 31;
	Kim.data.salary = 3000000;
	
	printf("Kim's age : %d \n", Kim.data.age);
	printf("Kim's salary : %d$/year \n", Kim.data.salary);
	
	return 0;
}
```
위 코드를 보면 다음과 같이 또다른 구조체 변수를 멤버로 가진 것을 볼 수 있다.
구조체는 사용자가 정의한 또 다른 타입이라고 볼 수 있기 때문에 이는 자연스럽다.

### 구조체 변수의 정의 방법
```c
#include <stdio.h>
char copy_str(char *dest, char *src);
int Print_Obj_Status(struct obj OBJ);
struct obj {
	char name[20];
	int x, y;
} Ball;

int main() {
	Ball.x = 3;
	Ball.y = 4;
	copy_str(Ball.name, "RED BALL");
	
	Print_Obj_Status(Ball);
	
	return 0;
}
int Print_Obj_Status(struct obj OBJ) {
	printf("Location of %s \n", OBJ.name);
	printf("( %d , %d ) \n", OBJ.x, OBJ.y);
	
	return 0;
}
char copy_str(char *dest, char *src) {
	while (*src) {
		*dest = *src;
		src++;
		dest++;
	}
	
	*dest = '\0';
	
	return 1;
}
```
위 구조체 정의를 보면 끝에 Ball이 추가된 것을 볼 수 있다.
이는 바로 그냥 `struct obj` 형의 `Ball`이라는 구조체 변수를 정의하라는 뜻이다.
우리가 `main` 함수 내부에서
```c
struct obj Ball;
```
이라고 써왔던 것과 다를 바가 없다. 그냥, 위와 같이 구조체 변수를 정의하는 방법도 있다.

### 더 쉽게 초기화
```c
#include <stdio.h>
int Print_Status(struct HUMAN human);
struct HUMAN {
	int age;
	int height;
	int weight;
	int gender;
};

int main() {
	struct HUMAN Adam = {31, 182, 75, 0};
	struct HUMAN Eve = {27, 166, 48, 1};
	
	Print_Status(Adam);
	Print_Status(Eve);
}

int Print_Status(struct HUMAN human) {
	if (human.gender == 0) {
		printf("MALE \n");
	} else {
		printf("FEMALE \n");
	}
	
	printf("AGE : %d / Height : %d / Weight : %d \n", human.age, human.height,
		 human.weight);
	
	if (human.gender == 0 && human.height >= 180) {
		printf("HE IS A WINNER!! \n");
	} else if (human.gender == 0 && human.height < 180) {
		printf("HE IS A LOSER!! \n");
	}
	
	printf("------------------------------------------- \n");
	
	return 0;
}
```
위 코드를 보면 다음과 같이 좀 더 쉽게 초기화할 수 있는 방법이 나와있다.
```c
struct HUMAN Adam = {31, 182, 75, 0};
struct HUMAN Eve = {27, 166, 48, 1};
```

또한 이렇게도 할 수 있다.
```c
struct HUMAN {
	int age;
	int height;
	int weight;
	int gender;
} Adam = {31, 182, 75, 0}, Eve = {27, 166, 48, 1};

struct obj {
  char name[20];
  int x, y;
} Ball = {"abc", 10, 2};
```
